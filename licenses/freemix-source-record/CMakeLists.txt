cmake_minimum_required(VERSION 3.28)
project(source-record-private C)
if(NOT FREEMIX_OBS_ROOT)
 message(FATAL_ERROR "Set FREEMIX_OBS_ROOT to the FreeMix OBS source tree with built libobs/frontend import libraries")
endif()
add_library(source-record MODULE source-record.c)
target_include_directories(source-record PRIVATE "${FREEMIX_OBS_ROOT}/libobs" "${FREEMIX_OBS_ROOT}/frontend/api" "${FREEMIX_OBS_ROOT}/build_2026/config" "${FREEMIX_OBS_ROOT}/deps/w32-pthreads" "${FREEMIX_OBS_ROOT}/.deps/obs-deps-2026-07-15-x64/include")
target_link_libraries(source-record PRIVATE "${FREEMIX_OBS_ROOT}/build_2026/libobs/RelWithDebInfo/obs.lib" "${FREEMIX_OBS_ROOT}/build_2026/frontend/api/RelWithDebInfo/obs-frontend-api.lib" "${FREEMIX_OBS_ROOT}/build_2026/deps/w32-pthreads/RelWithDebInfo/w32-pthreads.lib")
target_compile_definitions(source-record PRIVATE WIN32 _WINDOWS _CRT_SECURE_NO_WARNINGS)
set_target_properties(source-record PROPERTIES PREFIX "")
